case index @ (0 | 3) =>
  head.isDigit match
    case true if head > '1'                                        => (head :: tup._1, index + 1)
    case true if index == 0 && head == '1'                         => tup
    case false if index == 0 && List('+', '(', ' ').contains(head) => tup
    case false if index == 3 && List(')', ' ', '.').contains(head) => tup
    case _                                                         => (Nil, -100)
